cmake_minimum_required(VERSION 3.21)
project(TaskbarMonitor VERSION 1.0.2 LANGUAGES CXX RC)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Configure with an x64 MSVC toolchain on Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8;/EHsc>")

add_library(Foundation STATIC src/utils/Startup.cpp src/config/ConfigManager.cpp src/utils/Logger.cpp src/utils/SingleInstanceManager.cpp)
target_include_directories(Foundation PUBLIC src)
target_include_directories(Foundation SYSTEM PUBLIC third_party)
target_link_libraries(Foundation PUBLIC user32 shell32 ole32 advapi32)
configure_file(packaging/Version.rc.in Version.rc @ONLY)
set(app_sources src/ui/Settings.rc "${CMAKE_CURRENT_BINARY_DIR}/Version.rc" src/ui/SettingsWindow.cpp src/main.cpp src/core/Application.cpp src/taskbar/OverlayWindow.cpp src/taskbar/Placement.cpp src/ui/Renderer.cpp src/ui/MetricDisplay.cpp)
add_executable(TaskbarMonitor WIN32 ${app_sources})
add_executable(TaskbarMonitorPortable WIN32 ${app_sources})
target_compile_definitions(TaskbarMonitorPortable PRIVATE TASKBAR_PORTABLE)
set_target_properties(TaskbarMonitorPortable PROPERTIES OUTPUT_NAME TaskbarMonitor RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/portable")
add_library(Monitoring STATIC src/monitor/Metrics.cpp src/monitor/SystemCollectors.cpp
    src/monitor/GpuMonitor.cpp src/monitor/TemperatureMonitor.cpp src/monitor/PingMonitor.cpp
    src/monitor/MonitorEngine.cpp)
target_link_libraries(Monitoring PUBLIC Foundation iphlpapi ws2_32 pdh gdi32)
target_link_libraries(TaskbarMonitor PRIVATE Monitoring d2d1 dwrite)
target_link_libraries(TaskbarMonitorPortable PRIVATE Monitoring d2d1 dwrite)

include(CTest)
if(BUILD_TESTING)
  add_executable(DisplayTests tests/DisplayTests.cpp src/ui/Renderer.cpp src/ui/MetricDisplay.cpp src/taskbar/Placement.cpp)
  target_link_libraries(DisplayTests PRIVATE Monitoring d2d1 dwrite)
  add_test(NAME display COMMAND DisplayTests)
  add_executable(MonitorTests tests/MonitorTests.cpp)
  target_link_libraries(MonitorTests PRIVATE Monitoring)
  add_test(NAME monitoring COMMAND MonitorTests)
  add_executable(MonitorDiagnostics tests/MonitorDiagnostics.cpp)
  target_link_libraries(MonitorDiagnostics PRIVATE Monitoring)
  add_executable(PlacementTests tests/PlacementTests.cpp src/taskbar/Placement.cpp)
  target_include_directories(PlacementTests PRIVATE src)
  add_test(NAME placement COMMAND PlacementTests)
  add_executable(FoundationTests tests/FoundationTests.cpp)
  target_link_libraries(FoundationTests PRIVATE Foundation)
  add_test(NAME foundation COMMAND FoundationTests)
endif()
